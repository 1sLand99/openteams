{(() => {
  if (typeof window === 'undefined') return null;
  var path = window.location.pathname.replace(/\/+$/, '') || '/';
  var basePath = path === '/docs' || path.startsWith('/docs/') ? '/docs' : '';
  var relativePath = basePath ? (path.slice(basePath.length) || '/') : path;
  var localePrefixes = ['zh-Hans', 'zh-Hant', 'ja', 'ko', 'fr'];
  if (localePrefixes.some(function(prefix) {
    return relativePath === '/' + prefix || relativePath.startsWith('/' + prefix + '/');
  })) return null;
  try {
    if (document.referrer) {
      var referrerUrl = new URL(document.referrer);
      if (referrerUrl.origin === window.location.origin) return null;
    }
  } catch (error) {}
  var candidates = Array.isArray(navigator.languages) && navigator.languages.length
    ? navigator.languages
    : [navigator.language || 'en'];
  var matched = 'en';
  for (var i = 0; i < candidates.length; i += 1) {
    var normalized = String(candidates[i] || '').toLowerCase();
    if (normalized.indexOf('zh-hant') === 0 || normalized.indexOf('zh-tw') === 0 || normalized.indexOf('zh-hk') === 0 || normalized.indexOf('zh-mo') === 0) {
      matched = 'zh-Hant';
      break;
    }
    if (normalized.indexOf('zh') === 0) {
      matched = 'zh-Hans';
      break;
    }
    if (normalized.indexOf('ja') === 0) {
      matched = 'ja';
      break;
    }
    if (normalized.indexOf('ko') === 0) {
      matched = 'ko';
      break;
    }
    if (normalized.indexOf('fr') === 0) {
      matched = 'fr';
      break;
    }
  }
  if (matched === 'en') return null;
  var targetRelativePath = relativePath === '/' ? '/' + matched : '/' + matched + relativePath;
  var target = basePath + targetRelativePath + window.location.search + window.location.hash;
  if (target !== window.location.pathname + window.location.search + window.location.hash) {
    window.location.replace(target);
  }
  return null;
})()}

## Refresh agent models

openteams reads the model list reported by each agent CLI or ACP runtime. Model IDs cannot be added manually on the agent page because doing so would not add support to the underlying CLI.

<Steps>
<Step>
Open `Settings -> Agents`.

![Open agent settings](../images/en/agent_setting.png)
</Step>

<Step>
Select an agent, then click **Refresh models** beside the **Model** field. openteams queries the installed CLI or ACP runtime again and updates its model list.
</Step>

<Step>
Select an agent, then choose one of its discovered models from the **Model** field.
</Step>
</Steps>

The refresh button in the top-right corner refreshes the agent configuration, not the model list.

For openteams CLI and other provider-based runtimes, configure provider models under `Settings -> Service Providers`; the agent model list is synchronized from those providers.
